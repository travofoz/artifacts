3. LOCAL COPILOT AUTOMATION OVERRIDE SYSTEM

Monitors execution metrics for running background processes, translates them via the authenticated GitHub Copilot CLI interpreter tool, and maps high-performance dGPU parameters to the user registry paths.

import subprocess, psutil, winreg, json, sys

print("[*] Tracking system baseline process threads for 5 seconds...")
active_processes = set()
ignore_filters = {"svchost", "explorer", "conhost", "runtimebroker", "idle", "system", "lsass", "services"}

for _ in range(5):
    for process in psutil.process_iter(['name', 'exe']):
        try:
            name = process.info['name'].lower().replace(".exe", "")
            route = process.info['exe']
            if route and name not in ignore_filters: active_processes.add(route)
        except (psutil.NoSuchProcess, psutil.AccessDenied): continue
    subprocess.run(["timeout", "/t", "1"] if sys.platform == "win32" else ["sleep", "1"], shell=True)

ai_prompt = (
    f"Analyze these specific running paths:\n{chr(10).join(active_processes)}\n\n"
    "Identify which executables are gaming applications, broadcasting suites (OBS Studio/TikTok Studio), or AI training frameworks that require discrete dGPU hardware access. "
    "Return ONLY a valid string-based JSON array of the parsed file paths. Do not include markdown accents or conversation."
)

try:
    copilot_call = subprocess.run(["gh", "copilot", "explain", ai_prompt], capture_output=True, text=True, check=True)
    raw_output = copilot_call.stdout
    if "```json" in raw_output: raw_output = raw_output.split("```json")[1].split("```")[0].strip()
    elif "```" in raw_output: raw_output = raw_output.split("```")[1].split("```")[0].strip()
    target_routes = json.loads(raw_output.strip())
except Exception as e:
    print(f"[X] Execution failed. Ensure 'gh copilot' authentication state is valid. Error: {e}"); sys.exit()

print("\n========================= PROPOSED PATH REGISTRY MAP =========================")
for path in target_routes: print(f"[MAPPED TARGET] FORCE DISCRETE GPU TIER: {path}")

if input("\nCommit these DirectX performance modifications to user registry? (y/n): ").lower() == 'y':
    key_handle = winreg.CreateKeyEx(winreg.HKEY_CURRENT_USER, r"Software\Microsoft\DirectX\UserGpuPreferences", 0, winreg.KEY_SET_VALUE)
    for path in target_routes: winreg.SetValueEx(key_handle, path, 0, winreg.REG_SZ, "GpuPreference=2;")
    winreg.CloseKey(key_handle)
    print("[SUCCESS] Hardware parameters assigned successfully.")
